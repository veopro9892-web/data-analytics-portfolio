-- CTE Independent Aggregation eliminating the Fan-Out Trap
WITH gr_agg AS (
    SELECT po_number, line_item, SUM(received_quantity) AS total_received_qty
    FROM fact_goods_receipts
    GROUP BY po_number, line_item
),
inv_agg AS (
    SELECT po_number, line_item, 
           SUM(invoiced_quantity) AS total_invoiced_qty, 
           SUM(invoiced_amount) AS total_invoiced_amount,
           ROUND(SUM(invoiced_amount) / NULLIF(SUM(invoiced_quantity), 0), 2) AS avg_invoice_unit_price
    FROM fact_invoice_lines
    GROUP BY po_number, line_item
)
SELECT 
    pol.po_number,
    pol.line_item,
    s.supplier_name,
    pol.ordered_quantity,
    COALESCE(gr.total_received_qty, 0.0) AS received_quantity,
    COALESCE(inv.total_invoiced_qty, 0.0) AS invoiced_quantity,
    COALESCE(inv.avg_invoice_unit_price, 0.0) AS invoice_unit_price,
    CASE 
        WHEN inv.total_invoiced_qty > 0 AND gr.total_received_qty = 0 THEN 'EXCEPTION'
        WHEN inv.total_invoiced_qty > gr.total_received_qty THEN 'EXCEPTION'
        WHEN ABS((inv.avg_invoice_unit_price - pol.po_unit_price) / pol.po_unit_price) > 0.02 THEN 'EXCEPTION'
        WHEN pol.expected_delivery_date < '2026-11-17' AND gr.total_received_qty < pol.ordered_quantity THEN 'EXCEPTION'
        ELSE 'CLEAN_MATCH'
    END AS match_status
FROM fact_po_lines pol
JOIN fact_purchase_orders po ON pol.po_number = po.po_number
JOIN dim_suppliers s ON po.supplier_id = s.supplier_id
LEFT JOIN gr_agg gr ON pol.po_number = gr.po_number AND pol.line_item = gr.line_item
LEFT JOIN inv_agg inv ON pol.po_number = inv.po_number AND pol.line_item = inv.line_item;
